cmake_minimum_required(VERSION 3.20)
project(upe LANGUAGES CXX)
include(CTest)
add_library(parser_core STATIC src/source.cpp)
target_include_directories(parser_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(parser_core PUBLIC cxx_std_20)
function(add_focused_test target test_name source)
  add_executable(${target} EXCLUDE_FROM_ALL ${source})
  target_link_libraries(${target} PRIVATE parser_core)
  add_test(NAME ${test_name} COMMAND $<TARGET_FILE:${target}>)
endfunction()
if(BUILD_TESTING)
  add_focused_test(test_source_snapshot source.snapshot tests/test_source_snapshot.cpp)
  add_focused_test(test_source_slice source.slice tests/test_source_slice.cpp)
  add_focused_test(test_source_lines source.lines tests/test_source_lines.cpp)
endif()
