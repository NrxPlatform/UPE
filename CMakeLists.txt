cmake_minimum_required(VERSION 3.20)
project(upe LANGUAGES CXX)
include(CTest)
add_library(upe_source STATIC src/source.cpp src/encoding.cpp)
target_include_directories(upe_source PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(upe_source PUBLIC cxx_std_20)
add_library(upe_lexical STATIC src/lexical.cpp)
target_link_libraries(upe_lexical PUBLIC upe_source)
add_library(upe_diagnostics STATIC src/diagnostics.cpp)
target_link_libraries(upe_diagnostics PUBLIC upe_source)
# Backwards-compatible starter target.
add_library(parser_core ALIAS upe_source)
function(add_focused_test target test_name source library)
  add_executable(${target} EXCLUDE_FROM_ALL ${source})
  target_link_libraries(${target} PRIVATE ${library})
  add_test(NAME ${test_name} COMMAND $<TARGET_FILE:${target}>)
  add_dependencies(upe_tests ${target})
endfunction()
if(BUILD_TESTING)
  add_custom_target(upe_tests)
  add_focused_test(test_source_snapshot source.snapshot tests/test_source_snapshot.cpp upe_source)
  add_focused_test(test_source_slice source.slice tests/test_source_slice.cpp upe_source)
  add_focused_test(test_source_lines source.lines tests/test_source_lines.cpp upe_source)
  add_focused_test(test_source_encoding source.encoding tests/test_source_encoding.cpp upe_source)
  add_focused_test(test_source_unicode source.unicode tests/test_source_unicode.cpp upe_source)
  add_focused_test(test_token_range token.range tests/test_token_range.cpp upe_lexical)
  add_focused_test(test_token_trivia token.trivia tests/test_token_trivia.cpp upe_lexical)
  add_focused_test(test_token_synthetic token.synthetic tests/test_token_synthetic.cpp upe_lexical)
  add_focused_test(test_diagnostic diagnostic tests/test_diagnostic.cpp upe_diagnostics)
endif()
